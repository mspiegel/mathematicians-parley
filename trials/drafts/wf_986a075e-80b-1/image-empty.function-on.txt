import mundane theorem    stdlib/functions/value-in-image
import definition         stdlib/sets/subset
import mundane theorem    stdlib/sets/subset-empty

theorem image-empty
  note        F[B] has an element exactly when B has one, since each element
              of B goes to an element of F[B].
  let A be a set                                                      (H1)
  let F be a function on A                                            (H2)
  let B ⊆ A                                                           (H3)
  then F[B] = ∅ ↔ B = ∅

1.  F[B] = ∅ ↔ B = ∅
    both directions

    direction
    assume F[B] = ∅                                                   (D1)

    1.1.  For all t ∈ B, t ∈ ∅.
          proof
          let t ∈ B                                                   (K1)

          1.1.1.  F(t) ∈ F[B]
                  mun:value-in-image, from H2, H3, K1

          1.1.2.  F(t) ∈ ∅
                  substitute F[B] = ∅ (D1) into line 1.1.1

          1.1.3.  F(t) ∉ ∅
                  inspection, contradicting 1.1.2

    1.2.  B ⊆ ∅
          def:subset, from 1.1

    1.3.  B = ∅
          mun:subset-empty, from 1.2

    direction
    assume B = ∅                                                      (D2)

    1.4.  F[B] = ∅
          substitute B = ∅ (D2)
